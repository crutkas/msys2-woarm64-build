cmake_minimum_required(VERSION 3.20)
project(msys_staging_control LANGUAGES NONE)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "MSYS")
  message(FATAL_ERROR "This control requires the actual MSYS platform")
endif()
include(Platform/MSYS)
if(NOT CMAKE_SHARED_LIBRARY_PREFIX STREQUAL "msys-")
  message(FATAL_ERROR "The MSYS platform did not select its real DLL prefix")
endif()
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/payload.txt" DESTINATION share/staging-control)
