cmake_minimum_required(VERSION 3.20)
project(zstd_alias_control LANGUAGES NONE)
include(Platform/MSYS)
include("${ALIAS_HELPER}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/zstd.exe" "Unexecuted native-alias file fixture\n")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/zstd.1" "Manual alias fixture\n")
add_custom_target(zstd)
add_custom_target(zstd.1)
foreach(alias zstdcat unzstd zstdmt)
    zstd_add_alias("${alias}" zstd zstd alias_file)
    install(PROGRAMS "${alias_file}" DESTINATION bin)
endforeach()
foreach(alias zstdcat.1 unzstd.1)
    zstd_add_alias("${alias}" zstd.1 zstd.1 alias_file)
    install(FILES "${alias_file}" DESTINATION share/man/man1)
endforeach()
